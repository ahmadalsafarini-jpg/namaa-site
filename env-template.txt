# Firebase Configuration Template
# Copy this file to ".env" in your project root and fill in the values.
# Local:  After editing .env, restart "npm run dev".
# Vercel: Add the same variables in Project → Settings → Environment Variables (see VERCEL_ENV_SETUP.md), then redeploy.

# Firebase (from Firebase Console → Project settings → General → Your apps)
VITE_FIREBASE_API_KEY=your-api-key-here
VITE_FIREBASE_AUTH_DOMAIN=your-project-id.firebaseapp.com
VITE_FIREBASE_PROJECT_ID=your-project-id
VITE_FIREBASE_STORAGE_BUCKET=your-project-id.firebasestorage.app
VITE_FIREBASE_MESSAGING_SENDER_ID=your-sender-id
VITE_FIREBASE_APP_ID=your-app-id
VITE_FIREBASE_MEASUREMENT_ID=your-measurement-id
VITE_FIREBASE_DATABASE_URL=https://your-project-id-default-rtdb.europe-west1.firebasedatabase.app/

# Google Maps (Google Cloud Console → Credentials; enable Maps JavaScript API + Places API)
VITE_GOOGLE_MAPS_API_KEY=your-google-maps-api-key-here

# Admin portal password
VITE_ADMIN_PASSWORD=your-secure-admin-password-here

# Optional: backend for email notifications
VITE_BACKEND_URL=http://localhost:3001
